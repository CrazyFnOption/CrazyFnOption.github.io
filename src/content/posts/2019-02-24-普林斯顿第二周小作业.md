package Interview_Question_Week2.Elementary_Sorts;  
  
import edu.princeton.cs.algs4.StdRandom;  
  
enum Ball {  
    Red,  
    Blue,  
    White;  
}  
  
public class Backets  
{  
    private Ball backets[];  
  
    private Ball color(int i){  
        return backets[i];  
    }  
  
    private void swap (int i , int j){  
        Ball tmp = backets[i];  
        backets[i] = backets[j];  
        backets[j] = tmp;  
    }  
  
    public Backets(int n) {  
        backets = new Ball[n];  
        for (Ball i : backets){  
            int x = StdRandom.uniform(3);  
            switch (x){  
                case 0: {i = Ball.Blue;break;}  
                case 1: { i = Ball.Red;break;}  
                case 2: {i = Ball.White;break;}  
            }  
        }  
    }  
  
    public void sort() {  
        int low = 0;  
        int cur = 0;  
        int high = backets.length - 1;  
        while (cur <= high){  
            switch (color(cur)){  
                case Red: {  
                    if (cur != low) {  
                        swap(cur, low);  
                    }  
                    cur++;  
                    low++;  
                    break;  
                }  
  
                case White: {  
                    cur++;  
                    break;  
                }  
  
                case Blue: {  
                    swap (high,cur);  
                    high--;  
                    break;  
                }  
            }  
        }  
    }  
    public static void  main(String [] args){  
  
    }  
  
}

public class Intersection_of_two_sets {  
    private Set<Point> same;  
    private int sameTimes;  
  
    public Intersection_of_two_sets(Point[]a , Point[]b,int times){  
        same = new HashSet<Point>();  
        for (int i = 0;i < times;i++){  
            same.add(a[i]);  
            same.add(b[i]);  
        }  
        sameTimes = times * 2 - same.size();  
    }  
  
    public int get(){  
        return sameTimes;  
    }  
  
    public static void  main(String [] args){

package Interview_Question_Week2;  
  
import java.util.*;  
  
public class MaxArrayStack {  
    private int Max;  
    private int sz;  
    private int []a;  
  
    public MaxArrayStack(){  
        Max = -999999999;  
        a = new int [1];  
        sz = 0;  
    }  
  
    public void resize(int capacity){  
        int[] tmp = new int [capacity];  
        for (int i = 0;i != sz;i++){  
            tmp[i] = a[i];  
        }  
        a = tmp;  
    }  
  
    private void check(){  
        if (sz == a.length) resize(a.length * 2);  
    }  
  
    public void push(int item){  
        if (item > Max) Max = item;  
        check();  
        a[sz++] = item;  
    }  
  
    public int pop(){  
        return a[--sz];  
    }  
  
    public int getMax(){  
        return Max;  
    }  
  
    public String toString(){  
        return Arrays.toString(a);  
    }  
  
  
    public static void  main(String [] args){  
        MaxArrayStack mStack = new MaxArrayStack();  
        mStack.push(4);  
        mStack.push(5);  
        mStack.push(6);  
        mStack.push(2);  
        mStack.push(1);  
        mStack.push(10);  
        mStack.push(7);  
        mStack.push(7);  
  
        System.out.println(mStack);  
        System.out.println(mStack.getMax());  
  
    }  
}

package Interview_Question_Week2;  
  
import java.util.*;  
  
class MaxLinkedStack {  
    private int N;  
    private Node first;  
    private Node max;  
    private class Node {  
        private double item;  
        private Node next;  
    }  
    public MaxLinkedStack() {  
        N = 0;  
        first = null;  
        max = null;  
    }  
    public double getMax() {  
        return max.item;  
    }  
    public void push(double item) {  
        Node oldfirst = first;  
        first = new Node();  
        first.item = item;  
        first.next = oldfirst;  
        N++;  
        if (item >= getMax()) {  
            Node oldmax = max;  
            max = new Node();  
            max.next = oldmax;  
        }  
    }  
    public double pop() {  
        double tmp = first.item;  
        first = first.next;  
        N--;  
        if (tmp == getMax()) {  
            max = max.next;  
        }  
        return tmp;  
    }  
}

package Interview_Question_Week2;  
  
  
import java.util.*;  
  
  
public class StackQueue<Item> {  
    private Stack<Item> input = new Stack <Item>();  
    private Stack<Item> output = new Stack <Item>();  
  
    public int size (){  
        return input.size() + output.size();  
    }  
  
    public StackQueue() {  
  
    }  
  
    public boolean isEmpty() {  
        return size() == 0;  
    }  
  
    public void enqueue(Item item) {  
        if (item == null)  
            throw new IllegalArgumentException("wrong enquene");  
        input.push(item);  
    }  
  
    public Item dequeue() {  
        if (isEmpty()){  
            throw new IndexOutOfBoundsException("out of range");  
        }  
        if (output.isEmpty()){  
            while (!input.isEmpty()){  
                output.push(input.pop());  
            }  
        }  
        return output.pop();  
    }  
  
  
    public static void  main(String [] args){  
        StackQueue<Integer> squeue = new StackQueue<Integer>();  
        int i = 0;  
        int N = 100;  
        System.out.println("Size: " + squeue.size());  
        squeue.enqueue(i);  
        while (i <= N) {  
            if (i % 3 == 0) {  
                System.out.println("Dequeue: " + squeue.dequeue());  
            } else {  
                squeue.enqueue(i);  
                System.out.println("Enqueue: " + i);  
            }  
            ++i;  
        }  
        System.out.println("Size: " + squeue.size());  
        while (!squeue.isEmpty()) {  
            System.out.println("Dequeue: " + squeue.dequeue());  
        }  
        System.out.println("Size: " + squeue.size());  
    }  
  
}
