先贴出关于优先队列的代码  
**MaxPQ.java**

```java
package Interview_Question_Week4.PQ;

import edu.princeton.cs.algs4.StdIn;
import edu.princeton.cs.algs4.StdOut;

import java.util.Comparator;
import java.util.Iterator;
import java.util.NoSuchElementException;

public class MaxPQ<Key extends Comparable<Key>> implements Iterable<Key> {
    protected Key[] key;
    protected int len;
    private Comparator<Key> comparator;

    public MaxPQ(int init){
        key = (Key[]) new Object[1 + init];
        len = 0 + init;
    }

    public MaxPQ (int init , Comparator<Key> rule){
        comparator = rule;
        key = (Key[]) new Object[1 + init];
        len = 0 + init;
    }

    private boolean less(int i,int j){
        if (comparator == null){
            return key[i].compareTo(key[j]) < 0;
        }
        return comparator.compare(key[i],key[j]) < 0;
    }

    protected void change(int i,int j) {
        Key temp = key[i];
        key[i] = key[j];
        key[j] = temp;
    }

    protected void sink(int k){
        while (k * 2 <= len) {
            int j = k * 2;
            if (j < len && less(j,j+1)) j++;
            if (!less(k,j)) break;
            change(k,j);
            k = j;
        }
    }

    protected void up(int k){
        while (k > 1 && less(k / 2,k)) {
            change(k, k / 2);
            k = k / 2;
        }
    }

    private void resize(int n) {
        Key[] temp = (Key[]) new Object[n];
        for (int i = 1; i <= len;i++) {
            temp[i] = key[i];
        }
        key = temp;

    }

    public boolean isEmpty(){
        return len == 0;
    }

    public Key get(int n) {
        return key[n];
    }

    public int size() {
        return len;
    }

    public void insert(Key item) {
        if (item == null)
            throw new IllegalArgumentException("wrong");
        if (len + 1 == key.length)  resize(2 * key.length);
        key[++len] = item;
        up(len);
        assert isMaxHeap(1);
    }

    // 这里是帮助后面完成一个assert判断而存在的
    private boolean isMaxHeap(int k) {
        if (k > len) return true;
        int left = 2*k;
        int right = 2*k + 1;
        if (left  <= len && less(k, left))  return false;
        if (right <= len && less(k, right)) return false;
        return isMaxHeap(left) && isMaxHeap(right);
    }

    public Key delMax() {
        if (isEmpty())  throw new NoSuchElementException();
        Key item = key[1];
        key[1] = null;
        change(1,len--);
        sink(1);
        if (len + 1 <= key.length / 4 && len > 0 ) resize(key.length / 2);
        assert isMaxHeap(1);
        return item;
    }

    public Iterator<Key> iterator (){
        return new heapIterator();
    }

    private class heapIterator implements Iterator<Key>{
        private MaxPQ<Key> copy;

        public heapIterator() {
            if (comparator == null) {
                copy = new MaxPQ<Key>(len);
            }
            else copy = new MaxPQ<Key>(len,comparator);
            for (int i = 0;i < len;i++){
                copy.insert(key[i]);
            }
        }

        public boolean hasNext() {
            if (!isEmpty()) return true;
            return false;
        }

        public Key next() {
            if (!hasNext()) throw new NoSuchElementException();
            return copy.delMax();
        }

        @Override
        public void remove() {
            throw new UnsupportedOperationException();
        }
    }
    public static void main(String[] args) {
        edu.princeton.cs.algs4.MaxPQ<String> pq = new edu.princeton.cs.algs4.MaxPQ<String>();
        while (!StdIn.isEmpty()) {
            String item = StdIn.readString();
            if (!item.equals("-")) pq.insert(item);
            else if (!pq.isEmpty()) StdOut.print(pq.delMax() + " ");
        }
        StdOut.println("(" + pq.size() + " left on pq)");
    }

}
```

```java
package Interview_Question_Week4;

import Interview_Question_Week4.PQ.MaxPQ;
import edu.princeton.cs.algs4.StdRandom;

public class RandomPQ<Key extends Comparable<Key>> extends MaxPQ {

    private int len;
    public RandomPQ (int n) {
        super(n);
        this.len = super.len;
    }

    public Comparable<Key> sample() {
        int n = StdRandom.uniform(len + 1);
        return super.get(n);
    }

    public Comparable<Key> randomRemove () {
        int n = StdRandom.uniform(len + 1);
        Comparable<Key> item = key[n];
        change(n,len--);
        key[len + 1] = null;
        sink(n);
        return item;
    }

}
```

public class PriorityQueue {  
    private MaxPQ<Integer> left;  
    private MinPQ<Integer> right;  
  
    public PriorityQueue() {  
        left = new MaxPQ<Integer>();  
        right = new MinPQ<Integer>();  
    }  
  
    public double findMedian() {  
        int L = left.size();  
        int R = right.size();  
        if (L == R) return (left.max() + right.min()) / 2;  
        else if (L > R) return left.max();  
        else return right.min();  
    }  
  
    public void insert(int key) {  
        double median = findMedian();  
        int L = left.size();  
        int R = right.size();  
        if (key <= median) {  
            left.insert(key);  
            if (L - R > 1) {  
                right.insert(left.delMax());  
            }  
        }  
        else {  
            right.insert(key);  
            if (R - L > 1) {  
                left.insert(right.delMin());  
            }  
        }  
    }  
  
    public void removeMedian() {  
        int L = left.size();  
        int R = right.size();  
        if (L > R) {  
            left.delMax();  
        }  
        else {  
            right.delMin();  
        }  
    }  
}

> 有四个数 a,b,c,d 满足 a^3 + b^3 = c^3 +d^3，然后再给一个数 n，求出 n之内所有满足上述等式的四个数

**思路**  
此处就是运用优先队列，将没两个数的立方和加起来，然后再以立方和的大小来进行排序，作为最后排序的标准，最后进行比较，可做优化。

代码如下：

**Taxicab**

```java
package Interview_Question_Week4;

import edu.princeton.cs.algs4.MinPQ;
import edu.princeton.cs.algs4.StdOut;

public class Taxicab implements Comparable<Taxicab> {

    private final int a;
    private final int b;
    private final int cube;

    public Taxicab(int x,int y) {
        this.a = x;
        this.b = y;
        this.cube = x * x * x + y * y * y;
    }

    @Override
    public int compareTo(Taxicab a) {
        if (this.cube < a.cube) return -1;
        if (this.cube > a.cube) return 1;
        if (this.a < a.a) return -1;
        if (this.a > a.a) return 1;
        return 0;
    }

    @Override
    public String toString() {
        return "number: " + cube + " (" + a + ", " + b + ")";
    }

    public void findTaxicabNumber(int N) {
        MinPQ<Taxicab> candidate = new MinPQ<Taxicab>();
        for (int i = 1;i <= N;i++) {
            candidate.insert(new Taxicab(i,i));
        }
        int cnt = 1;
        Taxicab pre = new Taxicab(0,0);
        while (!candidate.isEmpty()) {
            Taxicab curr = candidate.delMin();
            if (curr == pre) {
                cnt ++;
                if (cnt == 2) {
                    StdOut.print(pre.cube + "=" + pre);
                }
                StdOut.print(" = " + curr);
            }
            else {
                if (cnt > 1) StdOut.println();
                cnt = 1;
            }
            pre = curr;
            if (curr.a < N) candidate.insert(new Taxicab(curr.a,curr.b + 1));
        }
    }
}
```
