#include<queue>  
#include<string>   
#include<stdio.h>   
using namespace std;   
int main(){  
    int n;   
    while(cin>>n){  
        queue<string> a;   
        for(int i=0;i<n;i++){  
            string name;   
            cin>>name;   
            a.push(name);  
        }  
    int w,s;  
    scanf("%d,%d",&w,&s);   
    for(int i=0;i<w-1;i++){  
        string temp = a.front();   
        a.pop();  
        a.push(temp);  
    }   
    while(a.size()){  
        for(int i=0;i<s-1;i++){  
            string temp = a.front();  
            a.pop()  
            a.push(temp);   
            }  
        cout<<a.front()<<endl;  
        a.pop();   
        }  
    }   
}
