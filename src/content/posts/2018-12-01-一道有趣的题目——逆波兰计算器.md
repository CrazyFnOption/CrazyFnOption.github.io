**题解**  
首先这道题毫无疑问是要用到stringstream流的。分别依次读取字符串，并且将其导入到char型的代码中去。比如说a+b可以变成ab+，这里最好运用到栈的知识，来控制出栈与进栈。  
==注意以下几个地方==

1. 注意输入`cin>>t`以后，需要用`getchar()`来清空上一个步骤留下来的回车符号
2. 注意此处需要严格进行判断栈内是否为空。
3. 这里是将乘除求模的运算放在最右边，这样就可以有优先级存在了
4. 括号会提前进行一步运算，然后消除掉左括号，或者左边的所有运算式，进行一个结算。
5. map这里存在的意义就在于 为判断进行提供条件。
6. 此道题目涉及字符串，栈，模拟的相应算法，需要有空的时候可以多看看。

#include<iostream>  
#include<string>  
#include<algorithm>  
#include<stack>  
#include<sstream>  
#include<map>  
  
using namespace std;  
  
stack<char>ope;  
stack<int>num;  
string str;  
map<char,int>m;  
  
void init(){  
    m['+']=m['-']=1;  
    m['*']=m['/']=m['%']=m['(']=m[')']=2;  
    while(!num.empty())  
        num.pop();  
    while(!ope.empty())  
        ope.pop();  
}  
//此处为基本的运算规模。  
void op1(int &a,int &b,const char &c){  
    if (c =='+')    a += b;  
    else if (c == '-') a=b-a;  
    else if (c == '*') a*=b;  
    else if (c == '/') a=b/a;  
    else if (c == '%') a=b%a;  
}  
//此处就是控制出栈的运算，一般就是括号内部的计算了。算出每一个括号内部的运算式。  
void op2(){  
    char ch = ope.top();  
    while (ch != '('&&!ope.empty()){  
        ope.pop();  
        int a = num.top();  
        num.pop();  
        int b = num.top();  
        num.pop();  
        op1(a,b,ch);  
        num.push(a);  
        if (!ope.empty())  
            ch = ope.top();  
    }//这个下面要注意的判断条件中一定要加上判断是否为空。  
  
    if (!ope.empty()&&ope.top() =='(')  
        ope.pop();  
}  
  
int main()  
{  
    int t;  
    cin>>t;  
    getchar();  
    while (t--){  
        getline(cin,str);  
        stringstream s(str);  
        init();  
        char tmp;  
        while (s>>tmp){  
            if(tmp>='0'&&tmp<='9'){  
                int x = 0;  
                do{  
                    if (m[tmp]) break;  
                    x *= 10;  
                    x += tmp - '0';//此处是将char转换成int  
                }while(s>>tmp);  
                num.push(x);  
            }  
            //注意这个地方的高明之处就在于碰见‘）’的返回，以及判断有没有‘（’，  
            比如连加的情况就把加减乘除的优先运算级给表示出来了。  
            //最后相当于将运算优先级高的全部放到了右边，然后有括号的就先解决括号，  
            及时左括号被提前弄没了也不要紧，存在有右括号就行，然后乘除永远放在加减的左边。  
              
            if (tmp==')') op2();  
            else if (m[tmp]==1){  
                if (!ope.empty()&&ope.top()!='(')  
                    op2();  
                ope.push(tmp);  
            }  
            else if (m[tmp]) ope.push(tmp);  
        }  
        int ans = num.top();  
        num.pop();  
        while (!num.empty()&&!ope.empty()){  
            op1(ans,num.top(),ope.top());  
            ope.pop();  
            num.pop();  
        }  
        cout << ans << endl;  
    }  
    return 0;  
}
