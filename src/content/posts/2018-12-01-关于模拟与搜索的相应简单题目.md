## 一道简单的dfs的题目

![](/2018/12/01/关于模拟与搜索的相应简单题目/15338976566985.jpg)

**代码如下：**

```c++
#include<iostream>
#include<algorithm>
#include<vector>

using namespace std;

const int maxn = 105;
char h[maxn][maxn];
int n,m,cnt=0;
void dfs(int i,int j){
    if (i<0||i>=n||h[i][j]!='W') return ;
    if (j<0||j>=m||h[i][j]!='W') return ;
    if (h[i][j]=='W'){
        h[i][j]='.';
        dfs(i-1,j);
        dfs(i-1,j-1);
        dfs(i+1,j);
        dfs(i+1,j+1);
        dfs(i,j-1);
        dfs(i-1,j+1);
        dfs(i,j+1);
        dfs(i+1,j-1);
    }
}
int main()
{
    int ans=0;
    while(cin>>n>>m&&(n!=0&&m!=0)){
        for (int i=0;i!=n;i++)
            for (int j=0;j!=m;j++){
                cin>>h[i][j];
            }
        for (int i=0;i!=n;i++){
            for (int j=0;j!=m;j++){
                if (h[i][j]=='W'){
                    dfs(i,j);
                    ans++;
                }
            }
        }
        cout<<ans<<endl;
        ans=0;
    }
    return 0;
}
```

题解简单的dfs搜索出每一种情况，然后让其返回就行了。

#include<stdio.h>  
#include<queue>  
using namespace std;  
  
int fxy[4][2]={{1,0},{-1,0},{0,1},{0,-1}};  
//朝着四个方向  
char dis[101][101];  
  
int kp[101][101];  
  
//以下的node就作为每一次前进的节点  
struct node{  
    int x,y,cnt;  
    node(int a=0,int b=0,int c=0){  
        x=a; y=b; cnt=c;  
    }  
};  
  
int main()  
{  
    int m,n,k,g,k1,g1;  
    while(~scanf("%d %d",&m,&n))  
    {  
        for(int i=0;i<m;i++)  
        {  
            scanf("%s",dis[i]);  
        }  
        //      用队列整合的思想来完成  
        queue<struct node>q;  
        for(int i=0;i<m;i++)  
            for(int j=0;j<n;j++){  
                kp[i][j]=-1;  
            }  
        //      两种初始化过程，并且找到起点与终点的所在位置  
        for(int i=0;i<m;i++)  
            for(int j=0;j<n;j++){  
                if(dis[i][j]=='S'){  
                    k=i;  
                    g=j;  
                }  
                if(dis[i][j]=='E'){  
                    k1=i;  
                    g1=j;  
                }  
            }  
          
        //        将起点存进去  
        node a(k,g,0);  
        q.push(a);  
        //        下面就是为bfs的相应搜索  
        while(!q.empty()){  
            struct node now=q.front();  
            q.pop();  
            kp[now.x][now.y]=now.cnt;  
            //            将起点走了多少步数以及起点的坐标点存入到题目中去  
            for(int i=0;i<4;i++){  
                node next;  
                next.x=now.x+fxy[i][0];  
                next.y=now.y+fxy[i][1];  
                //                判断走的下一步有没有出界，或者使下一步走的不会碰到路障  
                    if(next.x>=0&&next.x<m&&next.y>=0&&next.y<n&&dis[next.x][next.y]!='#'&&kp[next.x][next.y]==-1){  
                    node empt(next.x,next.y,now.cnt+1);  
                    q.push(empt);  
                }  
            }  
        }  
        if(kp[k1][g1]==-1) printf("Trapped!\n");  
        else printf("Escaped in %d minute(s).\n",kp[k1][g1]);  
    }  
    return 0;  
}

## 一道DFS与BFS的综合题

==需要多花时间来温习一遍==

**题目**  
![](/2018/12/01/关于模拟与搜索的相应简单题目/15346690956114.jpg)

**代码如下：**

```c++
#include<iostream>
#include<queue>
#include<algorithm>

using namespace std;

int n,m,k,cnt = 0;
int x_begin,x_end,y_begin,y_end;

const int maxn = 1005;

int dis[maxn][maxn];
char ch[maxn][maxn];

class Position{
public:
    int x;
    int y;
    int distance;
    
public:
    Position() = default;
    Position(int a, int b, int c):x(a),y(b),distance(c){};
    Position move(int i);
};

Position Position::move(int i){
    Position post = *this;
    if (i == 0)         post.x = this -> x + 1;
    else if (i == 1)    post.x = this -> x - 1;
    else if (i == 2)    post.y = this -> y + 1;
    else if (i == 3)    post.y = this -> y - 1;
    return post;
}

void bfs(){
    queue<Position>q_dis;
    Position p_begin (x_begin,y_begin,0);
    q_dis.push(p_begin);
    while(!q_dis.empty()){
        Position now = q_dis.front();
        q_dis.pop();
        dis[now.x][now.y]=now.distance;
        for(int i = 0; i != 4; i++){
            Position next;
            next = now.move(i);
            if (next.x >= 0 && next.x < n && next.y >= 0 && next.y < m &&
                dis[next.x][next.y] == -1 && ch[next.x][next.y] != '#'){
                Position empt (next.x,next.y,now.distance + 1);
                q_dis.push(empt);
            }
        }
    }
}

void dfs(int x,int y){
    if (x < 0 || x >= n || y < 0 || y >= m) return ;
    if (ch[x][y]=='L')  {
        cnt++;
        return;
    }
    if (dis[x][y] == -1) return;
    //  前面已经有dis可以表示每一步走的距离，所以只需要一步一步来看，这样的做法便可以完全规避障碍物，单单从路径上面考虑。
    if (dis[x][y]==dis[x+1][y]+1)   dfs(x+1,y);
    if (dis[x][y]==dis[x-1][y]+1)   dfs(x-1,y);
    if (dis[x][y]==dis[x][y+1]+1)   dfs(x,y+1);
    if (dis[x][y]==dis[x][y-1]+1)   dfs(x,y-1);
}

int main()
{
    int T,Case = 0;
    cin>>T;
    while(T--){
        Case++;
        cin>>n>>m>>k;
        bool f_begin = false,f_end = false;
        for (int i = 0; i != n; i++){
            for (int j = 0; j != m; j++){
                cin >> ch[i][j];
                dis[i][j] = -1;
            }
        }
        for (int i = 0; i != n; i++){
            for (int j = 0; j != m; j++){
                if (f_end && f_begin)   break;
                if (ch[i][j] == 'L'){
                    x_begin = i;
                    y_begin = j;
                    f_begin = true;
                }
                if (ch[i][j] == 'C'){
                    x_end = i;
                    y_end = j;
                    f_end = true;
                }
            }
        }
        bfs();
        if (dis[x_end][y_end] == -1 ||dis[x_end][y_end] > k )
            cout << "Case #"<<Case<<": "<<-1<<endl;
        else{
            dfs(x_end,y_end);
            cout<<"Case #"<<Case<<": "<<dis[x_end][y_end]<<" "<<cnt<<endl;
            cnt = 0;
        }
    }
    return 0;
}
```

这里注明一下关于==默认实参==的知识点，因为在这个地方可以直接对类的构造函数赋值一个默认实参，可以避免输入默认构造函数了。

下面则是一道DFS的题目，这道题目看上去可以用bfs来做

![-w1003](/2018/12/01/关于模拟与搜索的相应简单题目/15351254685296.jpg)

**代码如下：**

```c++
#include <iostream>
#include <algorithm>
#include <queue>

using namespace std;

const int maxn = 100;
int road[maxn][maxn];
int dp[maxn][maxn];
int n, m;

int dfs (int i,int j){
    if (dp[i][j] != -1)   return dp[i][j];
    int Max = 0;
    if (i-1>=0 && road[i][j]>road[i-1][j]&&Max<dfs(i-1,j))
        Max = dfs(i-1,j);
    if(i+1<n && road[i][j]>road[i+1][j]&&Max<dfs(i+1,j))
        Max=dfs(i+1,j);
    if(j-1>=0 && road[i][j]>road[i][j-1]&&Max<dfs(i,j-1))
        Max=dfs(i,j-1);
    if(j+1<m && road[i][j]>road[i][j+1]&&Max<dfs(i,j+1))
        Max=dfs(i,j+1);
    return dp[i][j]=Max+1;
}

int main()
{
    while (cin >> n >> m)
    {
        for (int i = 0; i != n; i++)
        {
            for (int j = 0; j != m; j++)
            {
                cin >> road[i][j];
                dp[i][j] = -1;
            }
        }
        int MAX = -1;
        for (int i = 0; i != n; i++)
        {
            for (int j = 0; j != m; j++)
            {
                MAX = max (dfs(i,j),MAX);
            }
        }
        cout << MAX << endl;
    }
    return 0;
}
```

**题解**

此题其实最重要的就在于这个题目不同于其他题目，这个题目找的是最长路径，还不是最短路径，一般来说，BFS找的是最短路径，DFS找的是最短路径的条数。  
而这里就是利用dfs的返回值（每走一步的最大值，意思就是从各个角度上）记录在dp中，将每一个位置都走到不能走为止，然后存储在dp中，最后找最大值的DP即可。  
不过自己可以试试遍历的其他方法。
