**CountInversions**

```java
package Interview_Question_Week3;

import java.util.Arrays;

public class CountInversions {
    private Comparable[] aux;
    private int inversion = 0;

    private boolean less (Comparable a,Comparable b){
        return a.compareTo(b) < 0;
    }

    public  int get_num(Comparable[] a) {
        aux = new Comparable[a.length];
        sort(a,0,a.length - 1);
        return inversion;
    }

    public void sort(Comparable[] a,int lo,int high){
        if (lo >= high) return;
        int mid = (lo + high) / 2;
        sort(a,lo,mid);
        sort(a,mid+1,high);
        merge(a,lo,mid,high);
    }

    public void merge(Comparable[] a,int lo,int mid,int high){
        int i = lo;
        int j = mid + 1;

        for (int k = lo; k <= high;k++){
            aux[k] = a[k];
        }

        for (int k = lo;k <= high;k++){
            if (i > mid) a[k] = aux[j++];
            else if (j > high) a[k] = aux[i++];
            else if (less (aux[i],aux[j])){
                // 这一步的目的就在于逆序对的个数，前面有多少个大于自己个数，就有多少个逆序对
                inversion += mid + 1 - i;
                a[k] = aux[i++];
            }
            else a[k] = aux[j++];
        }
    }

    public static void  main(String [] args){
        Integer []a = {1,5,3,7,2,8,6,4};

        System.out.println(new CountInversions().get_num(a));
        System.out.println(Arrays.toString(a));
    }
}
```

package Interview_Question_Week3;  
  
import java.util.Arrays;  
import java.util.Iterator;  
import java.util.ListIterator;  
  
public class LinkedMergedArrays <T extends Comparable<T>> implements Iterable<T>{  
    private Node first = null;  
    private Node last = null;  
    private int n = 0;  
  
    private class Node {  
        Node next;  
        T item;  
    }  
  
    private boolean less (Comparable a,Comparable b) {  
        return a.compareTo(b) < 0;  
    }  
  
    public Iterator<T> iterator() {  
        return new ListIterator();  
    }  
  
    private class ListIterator implements Iterator<T>{  
        private Node current = first;  
        public boolean hasNext() {  
            return current != null;  
        }  
  
        public T next() {  
            // TODO Auto-generated method stub  
            T t = current.item;  
            current = current.next;  
            return t;  
        }  
    }  
  
    public void add(T t){  
        Node node = new Node();  
        node.item = t;  
        node.next = null;  
        if(first == null && last == null){  
            first = node;  
            last = node;  
        }else if(first != null && first == last){  
            first.next = node;  
            last = node;  
        }else{  
            last.next = node;  
            last = node;  
        }  
        n++;  
    }  
  
    public String toString(){  
        Iterator<T> iter = iterator();  
        String ret = iter.next().toString();  
        while(iter.hasNext()){  
            ret += ", "+ iter.next().toString() ;  
        }  
        return ret;  
    }  
  
    //以下代码是第一种方式的归并 还有第二种形式的归并 明天可以写一下。  
  
    public void mergeSort(){  
        first = sort(first);  
    }  
  
    private Node sort(Node head){  
        if(head == null || head.next == null) return head;  
        Node slow = head;  
        Node fast = head;  
        //取中间节点  
        while(fast.next != null && fast.next.next != null){  
            slow = slow.next;  
            fast = fast.next.next;  
        }  
        Node left = head;  
        Node right = slow.next;  
        slow.next = null; //将左右链表分开  
        left = sort(left);  
        right = sort(right);  
        return merge(left,right);  
    }  
  
    private Node merge(Node left, Node right){  
        //System.out.println("left="+left.element+",right="+right.element);  
        Node aux = new Node(); //需要耗费logn的额外空间  
        Node l= left;  
        Node r = right;  
        Node current = aux;  
        while(l != null && r!=null){  
            if(less(r.item,l.item)) {  
                current.next = r;  
                current = current.next;  
                r = r.next;  
            }  
            else {  
                current.next = l;  
                current = current.next;  
                l= l.next;  
            }  
        }  
        if(l!=null) current.next = l; // 如果左侧没遍历完，将其连接至current后  
        else if(r != null) current.next = r; //如果右侧没遍历完，将其连接至current后  
        return aux.next; //返回归并好的链表  
    }  
}

```java
package Interview_Question_Week3;

import edu.princeton.cs.algs4.StdRandom;

public class MergeSortedSubArray {
    private static boolean less (Comparable a,Comparable b){
        return a.compareTo(b) < 0;
    }

    public static void merge(Comparable[] array){
        int n = array.length / 2;
        Comparable[] aux = new Comparable[n];
        for (int i = 0;i < n;i++){
            aux[i] = array[i];
        }

        System.out.println(Arrays.toString(aux));
        int l = 0;
        int r = n;
        for(int k = 0; k<2*n;k++){
            if(l >= n) break;//辅助元素数组全部用完，array右侧不需要挪动位置了
            else if(r>=2*n) array[k]=aux[l++];//array原右侧元素全部放置合适位置，后面只需把辅助数组的元素挪到array右侧
            else if(less(array[r],aux[l])) array[k] = array[r++];
            else array[k] = aux[l++];
        }
    }
}
```
